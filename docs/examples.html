Examples

33 statements. 19 in Postgres, 14 in Databricks. Every entry parses, resolves, round-trips, and projects to the product IR. A test enforces that.

The same list drives the example picker in the report helper. Open the app with make dev and pick an entry. The filter panel fills itself from the statement.

The hard shapes

One panel for any query

The filter panel is not a fixed form. It walks the product IR and builds one row per predicate. A literal value gets a typed control. A comparison between two columns becomes a read-only row, because a value patch would change the meaning.

Postgres (19)

Simple filter

postgreswhere
SELECT cust_id, status, amount
FROM sales.orders
WHERE status = 'paid'

Inner join with two predicates

postgresjoin
SELECT c.region, SUM(o.amount) AS total
FROM sales.orders o
JOIN sales.customers c ON c.id = o.cust_id AND c.active = TRUE
WHERE o.status = 'paid'
GROUP BY c.region

Three filter kinds in one WHERE

postgreswhereinbetween
SELECT id, cust_id, amount, region
FROM sales.orders
WHERE status = 'paid'
  AND amount BETWEEN 100 AND 500
  AND region IN ('EU', 'US')

Left join used as an anti join

postgresjoinnull
SELECT c.id, c.email
FROM sales.customers c
LEFT JOIN sales.orders o ON o.cust_id = c.id
WHERE o.id IS NULL

Three way join

postgresjoin
SELECT o.id, c.email, p.method, p.amount
FROM sales.orders o
JOIN sales.customers c ON c.id = o.cust_id
JOIN sales.payments p ON p.order_id = o.id AND p.amount > 0
WHERE c.region IN ('EU', 'US')

Self join for the previous order

postgresjoinwindow
SELECT curr.id, prev.id AS previous_id
FROM sales.orders curr
LEFT JOIN sales.orders prev
  ON prev.cust_id = curr.cust_id AND prev.created_at < curr.created_at
WHERE curr.status = 'open'

Correlated EXISTS

postgressubqueryexists
SELECT c.id, c.email
FROM sales.customers c
WHERE EXISTS (
  SELECT 1 FROM sales.orders o
  WHERE o.cust_id = c.id AND o.status = 'paid'
)

Correlated NOT EXISTS

postgressubqueryexists
SELECT c.id
FROM sales.customers c
WHERE NOT EXISTS (
  SELECT 1 FROM sales.orders o
  WHERE o.cust_id = c.id AND o.amount > 500
)

Scalar subquery in the projection

postgressubqueryaggregate
SELECT o.id, o.amount,
  (SELECT MAX(p.amount) FROM sales.payments p WHERE p.order_id = o.id) AS largest_payment
FROM sales.orders o
WHERE o.amount > 50

IN against a subquery

postgressubqueryin
SELECT c.id
FROM sales.customers c
WHERE c.region IN (
  SELECT r.code FROM sales.regions r WHERE r.active
)

Derived table with a filter

postgressubquery
SELECT t.cust_id, t.total
FROM (
  SELECT o.cust_id, SUM(o.amount) AS total
  FROM sales.orders o
  GROUP BY o.cust_id
) t
WHERE t.total > 1000

CTE feeding a join

postgresctejoin
WITH paid AS (
  SELECT o.cust_id, SUM(o.amount) AS total
  FROM sales.orders o
  WHERE o.status = 'paid'
  GROUP BY o.cust_id
)
SELECT c.region, COUNT(*) AS customers
FROM paid p
JOIN sales.customers c ON c.id = p.cust_id
GROUP BY c.region

Two CTEs, one joined to the other

postgresctejoin
WITH paid AS (
  SELECT cust_id, SUM(amount) AS total FROM sales.orders
  WHERE status = 'paid' GROUP BY cust_id
),
big AS (
  SELECT cust_id FROM paid WHERE total > 1000
)
SELECT c.id, c.email
FROM sales.customers c
JOIN big b ON b.cust_id = c.id
WHERE c.active

Recursive CTE over an org tree

postgrescterecursive
WITH RECURSIVE tree AS (
  SELECT id, manager_id, 1 AS depth
  FROM hr.employees
  WHERE manager_id IS NULL
  UNION ALL
  SELECT e.id, e.manager_id, t.depth + 1
  FROM hr.employees e
  JOIN tree t ON e.manager_id = t.id
)
SELECT id, depth FROM tree WHERE depth < 5

Window function with QUALIFY-free ranking

postgreswindowsubquery
SELECT id, cust_id, amount
FROM (
  SELECT id, cust_id, amount,
    ROW_NUMBER() OVER (PARTITION BY cust_id ORDER BY amount DESC) AS rn
  FROM sales.orders
) ranked
WHERE rn = 1

UNION of two branches

postgresset
SELECT cust_id FROM sales.orders WHERE status = 'paid'
UNION
SELECT cust_id FROM sales.refunds WHERE settled

BETWEEN, LIKE, and a parameter

postgreswhereparam
SELECT id, amount
FROM sales.orders
WHERE amount BETWEEN 10 AND 500
  AND email LIKE '%@example.com'
  AND status = $1

DISTINCT ON keeps the first row

postgres
SELECT DISTINCT ON (cust_id) cust_id, id, created_at
FROM sales.orders
ORDER BY cust_id, created_at DESC

LATERAL join to a top-three subquery

postgresjoinsubquery
SELECT c.id, top.id AS order_id, top.amount
FROM sales.customers c
JOIN LATERAL (
  SELECT o.id, o.amount FROM sales.orders o
  WHERE o.cust_id = c.id
  ORDER BY o.amount DESC
  LIMIT 3
) top ON TRUE
WHERE c.region = 'EU'

Databricks (14)

Simple filter

databrickswhere
SELECT cust_id, status, amount
FROM sales.orders
WHERE status = 'paid'

QUALIFY on a window function

databricksqualifywindow
SELECT city, car_model, quantity
FROM dealer
QUALIFY RANK() OVER (PARTITION BY car_model ORDER BY quantity) = 1

QUALIFY references an output alias

databricksqualify
SELECT city, car_model,
  RANK() OVER (PARTITION BY car_model ORDER BY quantity) AS rank
FROM dealer
QUALIFY rank = 1

SEMI join

databricksjoin
SELECT o.id, o.amount
FROM sales.orders o
LEFT SEMI JOIN sales.customers c ON c.id = o.cust_id
WHERE o.status = 'paid'

ANTI join

databricksjoin

FROM sales.customers c

CTE with a QUALIFY step

databricksctequalify
WITH recent AS (
  SELECT cust_id, id, amount, created_at
  FROM sales.orders
  WHERE created_at >= '2026-01-01'
)
SELECT cust_id, id, amount
FROM recent
QUALIFY ROW_NUMBER() OVER (PARTITION BY cust_id ORDER BY created_at DESC) = 1

Recursive CTE with MAX RECURSION

databrickscterecursive
WITH RECURSIVE t(n) AS (
  SELECT 1
  UNION ALL
  SELECT n + 1 FROM t WHERE n < 5
)
MAX RECURSION LEVEL 10
SELECT * FROM t

Three CTE layers

databrickscte
WITH base AS (
  SELECT cust_id, amount, status FROM sales.orders
),
paid AS (
  SELECT cust_id, SUM(amount) AS total FROM base WHERE status = 'paid' GROUP BY cust_id
),
top AS (
  SELECT cust_id, total FROM paid WHERE total > 500
)
SELECT c.id, top.total
FROM sales.customers c
JOIN top ON top.cust_id = c.id

EdgeSQL started from one observation: changing SQL with string replacement breaks as soon as two edits touch the same name. A typed tree with a preview gate is the smallest honest fix, so he built it and tested it against the official Postgres and Databricks statements.
